        // Process query parameters and redirect to SearchResults
        (function() {
            var searchParams = new URLSearchParams(window.location.search);
            
            // Verify that we have the necessary parameters
            if (!searchParams.has('search') || !searchParams.has('Rec_Book') || !searchParams.has('Rec_Page')) {
                // Redirect to homepage if parameters are missing
                redirectToHome();
                return;
            }

            var bookNumber = searchParams.get('Rec_Book');
            var pageNumber = searchParams.get('Rec_Page');
            
            // Determine the base URL for the application
            var baseUrl = window.location.pathname.includes('/officialrecords') 
                ? '/officialrecords' 
                : '';
            
            console.log('Starting search with Book:', bookNumber, 'Page:', pageNumber);
            
            // Function to redirect to home in case of error
            function redirectToHome() {
                console.log('Redirecting to home...');
                window.location.href = `${baseUrl}`;
            }
            
            // This legacy redirect always skips server-side verification (skipCaptcha=true),
            // so no verification token needs to be obtained here.
            async function executeSearch() {
                try {
                    // Make API call
                    console.log('Making API call...');
                    var apiUrl = `${window.location.origin}${baseUrl}/api/home/recordingsearch?bookNumber=${bookNumber}&pageNumber=${pageNumber}&bookType=&searchType=Recording Book/Page&token=&skipCaptcha=true`;
                    
                    var response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: {
                            'Accept': 'application/json',
                            'Content-Type': 'application/json; charset=utf-8'
                        }
                    });

                    if (!response.ok) {
                        throw new Error(`API call failed with status: ${response.status}`);
                    }

                    var data = await response.json();
                    console.log('API response received:', data);
                    
                    if (data && data.qs) {
                        console.log('Redirecting to SearchResults...');
                        window.location.href = `${baseUrl}/SearchResults?qs=${encodeURIComponent(data.qs)}`;
                    } else {
                        throw new Error('No query string received in response');
                    }
                    
                } catch (error) {
                    console.error('Error in executeSearch:', error);
                    // Redirect to homepage if something fails
                    redirectToHome();
                }
            }
            
            // Execute the search
            executeSearch();
            
        })();
    </script>
